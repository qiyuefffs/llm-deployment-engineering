import base64
import requests
import json

def run_image_stream_inference(image_path):
    # -------- 配置 --------
    API_URL = "http://<NODE_IP>:8007/v1/chat/completions"
    MODEL_NAME = "qwen3-vl-8b-instruct"

    # -------- 读取图片并转 base64 --------
    try:
        with open(image_path, "rb") as f:
            image_bytes = f.read()
    except FileNotFoundError:
        print(f"[错误] 找不到文件: {image_path}")
        return
    except Exception as e:
        print("[错误] 打开图片失败:", e)
        return

    image_base64 = base64.b64encode(image_bytes).decode("utf-8")

    # -------- 构造请求体 --------
    payload = {
        "model": MODEL_NAME,
        "messages": [
            {
                "role": "user",
                "content": [
                    {
                        "type": "text",
                        "text": "请描述这张图片的内容，并指出图片中的重点元素。"
                    },
                    {
                        "type": "image_url",
                        "image_url": {
                            "url": f"data:image/jpeg;base64,{image_base64}"
                        }
                    }
                ]
            }
        ],
        "stream": True   # ⭐ 开启流式输出
    }

    headers = {
        "Content-Type": "application/json"
    }

    # -------- 发送请求 (流式) --------
    print("=== 请求已发送，开始接收流式输出... ===")
    try:
        with requests.post(API_URL, headers=headers, data=json.dumps(payload), stream=True, timeout=300) as response:
            print("HTTP Status:", response.status_code)

            # 如果状态不是 200，则直接打印错误
            if response.status_code != 200:
                print("[错误] 服务端返回:")
                print(response.text)
                return

            # 一行一行读取 SSE / chunk 流
            for line in response.iter_lines(decode_unicode=True):
                if not line:
                    continue

                try:
                    text = line.strip()
                    # 如果是 SSE 的 data: 开头
                    if text.startswith("data: "):
                        text = text[len("data: "):]

                    # 检查是否结束标记
                    if text == "[DONE]":
                        print("\n=== 流式输出结束 ===")
                        break

                    # 解析 JSON
                    chunk = json.loads(text)
                    if "choices" in chunk:
                        for c in chunk["choices"]:
                            delta = c.get("delta")
                            if delta and "content" in delta:
                                print(delta["content"], end="", flush=True)

                except json.JSONDecodeError:
                    # 不是 JSON 的部分也打印出来
                    print(line)

    except Exception as e:
        print("[错误] 网络请求过程中出现异常:", e)


if __name__ == "__main__":
    print("=== Qwen3-VL-8B-Instruct 图片解析 流式推理 ===")
    img_path = input("请输入要解析的本地图片路径: ").strip()
    run_image_stream_inference(img_path)
