# R29 · AMD单节点四卡DS-70B-fp16测试命令
并发数1

```text
docker run -it --name deepseek-vllm-benchmark-4-1 \
  --device=/dev/kfd \
  --device=/dev/dri \
  --group-add=video \
  --ipc=host \
  --shm-size=16g \
  -v /root/modles:/app/models \
  -v /root/results:/app/vllm/benchmarks/results \
  -e HSA_OVERRIDE_GFX_VERSION=11.0.0 \
  -e VLLM_WORKER_MULTIPROC_METHOD=spawn \
  ghcr.io/embeddedllm/vllm-rocm:v0.8.2-18ed313 \
  bash -c "python3 /app/vllm/benchmarks/benchmark_throughput.py \
    --backend vllm \
    --model /app/models/DeepSeek-R1-Distill-Llama-70B \
    --tokenizer /app/models/DeepSeek-R1-Distill-Llama-70B \
    --tensor-parallel-size 4 \
    --data-parallel-size 1 \
    --max-model-len 4096 \
    --num-prompts 128 \
    --input-len 32 \
    --output-len 128 \
    --dtype auto \
    --gpu-memory-utilization 0.85 \
    --disable-custom-all-reduce \
    --max-num-seqs 1 \
    | tee /app/vllm/benchmarks/results/benchmark_output_tp_4_dp_1_c1.txt"

```
